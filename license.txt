protected void GVCategories_RowDeleting(object sender, GridViewDeleteEventArgs e)
{
    int no = 0;
    try
    {
        if (conn.State == ConnectionState.Closed)
        {
            conn.Open();

            cmd = new MySqlCommand();
            cmd.Connection = conn;
            cmd.CommandType = CommandType.Text;
            cmd.CommandText = "DELETE FROM category WHERE id=@id";

            int id = Convert.ToInt32(GVCategories.DataKeys[e.RowIndex].Value);

            cmd.Parameters.Clear();
            cmd.Parameters.AddWithValue("@id", id);

            no = cmd.ExecuteNonQuery();
        }
    }
    catch (Exception ex)
    {
        Response.Write("GVCategories_RowDeleting: " + ex.Message);
    }
    finally
    {
        if (conn.State == ConnectionState.Open)
        {
            conn.Close();
        }

        if (no > 0)
        {
            FillGVCategories();
        }
    }

}

protected void GVCategories_RowEditing(object sender, GridViewEditEventArgs e)
{
    GVCategories.EditIndex = e.NewEditIndex;
    FillGVCategories();
}

protected void GVCategories_RowUpdating(object sender, GridViewUpdateEventArgs e)
{
    int id = Convert.ToInt32(GVCategories.DataKeys[e.RowIndex].Value);

    TextBox textBox = (TextBox)GVCategories.Rows[e.RowIndex].Cells[0].Controls[0];
    string newCategory = textBox.Text;

    if (conn.State == ConnectionState.Closed)
    {
        conn.Open();

        cmd = new MySqlCommand();
        cmd.Connection = conn;
        cmd.CommandType = CommandType.Text;
        cmd.CommandText = "UPDATE category SET category_name=@category_name WHERE id=@id;";
        cmd.Parameters.Clear();
        cmd.Parameters.AddWithValue("@category_name", newCategory);
        cmd.Parameters.AddWithValue("@id", id);
        cmd.ExecuteNonQuery();

        GVCategories.EditIndex = -1;
        FillGVCategories();
    }
}

protected void GVCategories_RowCancelingEdit(object sender, GridViewCancelEditEventArgs e)
{
    GVCategories.EditIndex = -1;
    FillGVCategories();
}
